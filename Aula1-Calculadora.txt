package calculadora;

/**
 *
 * @author Aluno CA
 */
public class Operacao {
    int maisT;
    int menosT;
    int multiT;
    float diviT;
    
    public int soma (int a, int b){
          
        maisT = a + b;
        return maisT;
    }
    public void imprimirAd() {
        System.out.println("Valor da soma: " + maisT);
    }
        
    public int subtracao (int a, int b) {
        menosT = a - b;
        return menosT;
    }
    public void imprimirSb() {
         System.out.println("Valor da subtração: " + menosT);
    }
    
    public int multiplicacao(int a, int b) {
        multiT = a * b;
        return multiT;
    }
    public void imprimirMt(){
        System.out.println("valor da multiplicação: " + multiT);
    }
    
    public float divisao (float a, float b){
        diviT = a / b;
        return diviT;
    }
    public void imprimirDv (){
        System.out.println("Valor da divisão: " + diviT);
    }
}
